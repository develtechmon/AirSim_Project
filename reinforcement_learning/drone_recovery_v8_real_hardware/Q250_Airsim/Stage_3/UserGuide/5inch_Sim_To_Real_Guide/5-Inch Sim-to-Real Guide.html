    class ArduCopterParams : public MultiRotorParams
    {
    public:
        ArduCopterParams(const AirSimSettings::MavLinkVehicleSetting& vehicle_setting, std::shared_ptr<const SensorFactory> sensor_factory)
            : sensor_factory_(sensor_factory)
        {
            connection_info_ = getConnectionInfo(vehicle_setting);
        }

        virtual ~ArduCopterParams() = default;

        virtual std::unique_ptr<MultirotorApiBase> createMultirotorApi() override
        {
            return std::unique_ptr<MultirotorApiBase>(new ArduCopterApi(this, connection_info_));
        }

    protected:
        virtual void setupParams() override
        {
            auto& params = getParams();

            // "Model" in settings.json selects the drone (default model name is "Generic")
            if (connection_info_.model == "Q250") {
                setupFrameQ250(params); // 5-inch drone, numbers from settings.json "Parameters"
            }
            else {
                setupFrameGenericQuad(params); // F450, unchanged (all Stage 1-3 training)
            }
        }

        // 5-inch drone. Same structure as setupFrameGenericQuad() in MultiRotorParams.hpp,
        // but every number comes from settings.json "Parameters" (or the default below).
        void setupFrameQ250(Params& params)
        {
            params.rotor_count = 4;
            std::vector<real_T> arm_lengths(params.rotor_count, q250Value("Q250_ARM_M", 0.125f));

            params.mass = q250Value("Q250_MASS_KG", 0.55f);
            real_T motor_assembly_weight = q250Value("Q250_MOTOR_KG", 0.033f);
            real_T box_mass = params.mass - params.rotor_count * motor_assembly_weight;

            // rotor: prop size, motor delay, and max thrust of one motor
            params.rotor_params.propeller_diameter = q250Value("Q250_PROP_D_M", 0.127f);
            params.rotor_params.control_signal_filter_tc = q250Value("Q250_MOTOR_TC_S", 0.02f);
            // choose max_rpm so that calculateMaxThrust() gives exactly Q250_THRUST_N
            // (thrust = C_T * air_density * (rpm/60)^2 * D^4, AirSim's own formula in RotorParams.hpp)
            const real_T thrust_n = q250Value("Q250_THRUST_N", 5.4f);
            const real_T d4 = static_cast<real_T>(std::pow(params.rotor_params.propeller_diameter, 4));
            const real_T rev_per_s = std::sqrt(thrust_n / (params.rotor_params.C_T * params.rotor_params.air_density * d4));
            params.rotor_params.max_rpm = rev_per_s * 60.0f;
            params.rotor_params.calculateMaxThrust();

            // centre block (stack + battery + Pi) as a box
            params.body_box.x() = q250Value("Q250_BODY_X_M", 0.15f);
            params.body_box.y() = q250Value("Q250_BODY_Y_M", 0.05f);
            params.body_box.z() = q250Value("Q250_BODY_Z_M", 0.07f);
            real_T rotor_z = 2.5f / 100; // same as the F450

            initializeRotorQuadX(params.rotor_poses, params.rotor_count, arm_lengths.data(), rotor_z);
            computeInertiaMatrix(params.inertia, params.body_box, params.rotor_poses, box_mass, motor_assembly_weight);
        }

        // value from settings.json "Parameters", or the default if the key is missing
        real_T q250Value(const std::string& key, real_T default_value) const
        {
            auto it = connection_info_.params.find(key);
            return (it != connection_info_.params.end()) ? static_cast<real_T>(it->second) : default_value;
        }

        virtual const SensorFactory* getSensorFactory() const override
        {
            return sensor_factory_.get();
        }

        static const AirSimSettings::MavLinkConnectionInfo& getConnectionInfo(const AirSimSettings::MavLinkVehicleSetting& vehicle_setting)
        {
            return vehicle_setting.connection_info;
        }

    private:
        AirSimSettings::MavLinkConnectionInfo connection_info_;
        std::shared_ptr<const SensorFactory> sensor_factory_;
    };
